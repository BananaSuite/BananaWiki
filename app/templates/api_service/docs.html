Python

import requests

BASE_URL = "{{ url_for('api_v1_status', _external=True).rstrip('/api/v1/status') }}"
TOKEN = "your_api_token_here"
HEADERS = {"Authorization": f"Bearer {TOKEN}", "Content-Type": "application/json"}

# List pages
resp = requests.get(f"{BASE_URL}/api/v1/pages", headers=HEADERS)
print(resp.json())

# Create a page
resp = requests.post(f"{BASE_URL}/api/v1/pages", headers=HEADERS, json={
    "title": "My New Page",
    "content": "# Hello World\n\nThis page was created via the API.",
    "slug": "my-new-page",
})
print(resp.json())

# Bulk create users
resp = requests.post(f"{BASE_URL}/api/v1/users/bulk", headers=HEADERS, json={
    "users": [
        {"username": "alice", "password": "securepass1", "role": "editor"},
        {"username": "bob", "password": "securepass2", "role": "user"},
    ]
})
print(resp.json())

# Bulk edit pages
resp = requests.post(f"{BASE_URL}/api/v1/pages/bulk-edit", headers=HEADERS, json={
    "edits": [
        {"slug": "my-new-page", "title": "Updated Title"},
    ]
})
print(resp.json())

# Get audit log
resp = requests.get(f"{BASE_URL}/api/v1/admin/audit-log?limit=10", headers=HEADERS)
print(resp.json())

JavaScript (Fetch)

const BASE = "{{ url_for('api_v1_status', _external=True).rstrip('/api/v1/status') }}";
const TOKEN = "your_api_token_here";

async function apiCall(method, path, body = null) {
  const resp = await fetch(`${BASE}${path}`, {
    method,
    headers: {
      "Authorization": `Bearer ${TOKEN}`,
      "Content-Type": "application/json",
    },
    body: body ? JSON.stringify(body) : null,
  });
  return resp.json();
}

// List users
apiCall("GET", "/api/v1/users").then(console.log);

// Create a page
apiCall("POST", "/api/v1/pages", {
  title: "JS Page",
  content: "# Created via JavaScript",
  slug: "js-page",
}).then(console.log);
